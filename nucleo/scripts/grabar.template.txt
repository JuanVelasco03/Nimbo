# Un ID de stream o nombre de modelo por linea. '*' = grabar todo.
# Los IDs que se van detectando aparecen en activos.txt y en la consola.
# Agregar una linea  -> empieza a grabar (se aplica en ~2s, sin reiniciar).
# Borrar esa linea   -> cierra la grabacion y genera el mp4 en grabaciones/.
# Se pueden poner varios modelos a la vez.

# marca; el nombre tiene que estar igual arriba como linea suelta.
@vpn=

# --- Calidad -------------------------------------------------------------
# max = la mejor que ofrezca cada modelo, sea 1080p, 720p o 480p.
# Se puede fijar igual:  @calidad=1080p,720p    @codec=av1,h264
@calidad=max
@codec=max
# 1 = le esconde las calidades bajas al navegador: en el master .m3u8 solo
#     le queda la emision original, asi el player no tiene a donde bajar.
#     Se aplica a cualquier stream que ofrezca la original, aunque el
#     nombre de la modelo todavia no se haya detectado (el master se pide
#     una sola vez, al arrancar el video: si se esperaba, no llegaba).
@forzar=1
# 1 = una sola variante por stream (la mejor). 'todas' = graba todas a la vez.
@variantes=1
# segundos para descubrir que calidades ofrece el stream antes de elegir
@espera=6
# si el reproductor cambia de calidad igual: segundos antes de seguirlo
# (se corta en otro archivo; lo de esos segundos no se pierde, va a buffer)
@cambio=8
# 1 = graba tambien el stream censurado de los shows privados (no es una
#     calidad mas baja: es otro video)
@borrosas=0
# Cuando la modelo se va a privado, el sitio sigue mandando video pero de
# 160p (un relleno, no se la ve). A esa altura o menos: corta la grabacion,
# guarda el mp4 en el acto y se queda esperando; cuando sale del privado
# vuelve a grabar sola, en un archivo nuevo. 0 = desactivado (lo graba).
@privado=160p
# 1 = si no hay ninguna calidad de las pedidas, graba la mejor que haya
# 0 = no graba nada hasta que vuelva la calidad pedida
@fallback=1

# --- Camino directo (sin navegador) --------------------------------------
# @directo=1 graba TODO lo de arriba sin abrirle pestana al supervisor:
# un hilo propio (grabador_directo.py) baja los segmentos de doppiocdn
# directo, descifrando el protocolo MOUFLON (ver specs\grabador_directo.md).
# El navegador queda de respaldo, pero el salto NO es automatico: si el
# protocolo deja de andar (stripchat rota las claves), esa modelo queda
# "degradada" (se ve en la pestana Registro) hasta que la agregues a
# @navegador= a mano -- grabar.txt lo edita solo Nimbo, nunca el grabador.
# @navegador=fulana,mengana opta por el navegador para esos nombres
# puntuales aunque @directo=1 este encendido arriba.
@directo=0
@navegador=
